enum class LimitStatus { Ok, Warning, Over, NoLimit }

fun limitStatus(spent: Long, limit: Long?): LimitStatus =
  when {
    limit == null -> LimitStatus.NoLimit
    spent >= limit -> LimitStatus.Over   // 100%+
    spent * 100 >= limit * 80 -> LimitStatus.Warning
    else -> LimitStatus.Ok
  }

Theme.kt

@Composable
fun BudzetTheme(
  dark: Boolean = isSystemInDarkTheme(),
  content: @Composable () -> Unit,
) {
  // kasnije: dynamicLight/DarkColorScheme(ctx)
  val colors = if (dark) DarkColors else LightColors
  val status = if (dark) DarkStatus else LightStatus
  CompositionLocalProvider(LocalStatus provides status) {
    MaterialTheme(
      colorScheme = colors,
      typography = BudzetType,
      shapes = BudzetShapes,
      content = content,
    )
  }
}

// Navigacija: skala fonta za labele max 1.3x
@Composable
fun CappedLabel(text: String) {
  val d = LocalDensity.current
  val capped = Density(d.density,
    d.fontScale.coerceAtMost(1.3f))
  CompositionLocalProvider(LocalDensity provides capped) {
    Text(text, maxLines = 1,
      overflow = TextOverflow.Ellipsis)
  }
}
